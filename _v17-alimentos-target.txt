/* ------------------------------------------- alertas e medidas fiscais */
function juntaComplementos(produtos, bases, frios, cnpjDireto) {
  var registros = {}, cnpjs = {}, processos = {};
  (produtos || []).forEach(function (p) {
    if (p.registro) registros[D.so(p.registro)] = 1;
    if (p.cnpj) cnpjs[D.so(p.cnpj)] = 1;
    if (p.processo) processos[D.so(p.processo)] = 1;
  });
  if (cnpjDireto) cnpjs[D.so(cnpjDireto)] = 1;

  var tarefas = [];
  Object.keys(cnpjs).forEach(function (c) {
    tarefas.push(alertasPorCnpj(c, bases, frios));
    tarefas.push(irregularPorCnpj(c, bases, frios));
  });
  Object.keys(registros).forEach(function (r) { tarefas.push(alertasPorRegistro(r, bases, frios)); });
  Object.keys(processos).forEach(function (p) { tarefas.push(irregularPorProcesso(p, bases, frios)); });

  return Promise.all(tarefas).then(function (rs) {
    var alertas = [], irregulares = [], vistos = {};
    rs.forEach(function (x) {
      (x && x.alertas || []).forEach(function (a) {
        if (!vistos['a' + a.a]) { vistos['a' + a.a] = 1; alertas.push(a); }
      });
      (x && x.irregulares || []).forEach(function (i) {
        var k = 'i' + (i.processo || i.id_dossie);
        if (!vistos[k]) { vistos[k] = 1; irregulares.push(i); }
      });
    });
    return { produtos: produtos, alertas: alertas, irregulares: irregulares };
  });
}

function alertasPorCnpj(cnpj, bases, frios) {
  var f = D.CHAVE.cnpj(cnpj);
  bases['alertas_sanitarios'] = 1;
  return D.busca('alertas_sanitarios/indices/cnpj/' + f + '.json').then(function (r) {
    if (r.semRede) frios.push(r);
    return { alertas: (r.dados || {})[cnpj] || [] };
  }).catch(function () { return { alertas: [] }; });
}

function alertasPorRegistro(reg, bases, frios) {
  var f = String(reg).slice(0, 3);
  bases['alertas_sanitarios'] = 1;
  return D.busca('alertas_sanitarios/indices/registros/' + f + '.json').then(function (r) {
    if (r.semRede) frios.push(r);
    return { alertas: (r.dados || {})[reg] || [] };
  }).catch(function () { return { alertas: [] }; });
}

function irregularPorCnpj(cnpj, bases, frios) {
  var f = String(cnpj || '').slice(0, 3);
  if (!f) return Promise.resolve({ irregulares: [] });
  bases['produtos_irregulares'] = 1;
  return D.busca('indices/irregulares_cnpj/' + f + '.json').then(function (r) {
    if (r.semRede) frios.push(r);
    var processos = (r.dados || {})[cnpj] || [];
    return Promise.all(processos.slice(0, 80).map(function (proc) {
      return irregularPorProcesso(proc, bases, frios);
    })).then(function (rs) {
      var itens = [];
      rs.forEach(function (x) { (x && x.irregulares || []).forEach(function (i) { itens.push(i); }); });
      return { irregulares: itens };
    });
  }).catch(function () { return { irregulares: [] }; });
}

function irregularPorProcesso(proc, bases, frios) {
  var f = D.CHAVE.irregulares(proc);
  if (!f) return Promise.resolve({ irregulares: [] });
  bases['produtos_irregulares'] = 1;
  return D.busca('produtos_irregulares/' + f + '.json').then(function (r) {
    if (r.semRede) frios.push(r);
    var d = (r.dados || {})[proc];
    return { irregulares: d ? [Object.assign({ processo: proc }, d)] : [] };
  }).catch(function () { return { irregulares: [] }; });
}

function alertasPorCnpj(cnpj, bases, frios) {
  var f = D.CHAVE.cnpj(cnpj);
  bases['alertas_sanitarios'] = 1;
  return D.busca('alertas_sanitarios/indices/cnpj/' + f + '.json').then(function (r) {
    if (r.semRede) frios.push(r);
    return { alertas: (r.dados || {})[cnpj] || [] };
  }).catch(function () { return { alertas: [] }; });
}

function alertasPorRegistro(reg, bases, frios) {
  var f = String(reg).slice(0, 3);
  bases['alertas_sanitarios'] = 1;
  return D.busca('alertas_sanitarios/indices/registros/' + f + '.json').then(function (r) {
    if (r.semRede) frios.push(r);
    return { alertas: (r.dados || {})[reg] || [] };
  }).catch(function () { return { alertas: [] }; });
}

function irregularPorCnpj(cnpj, bases, frios) {
  var f = String(cnpj || '').slice(0, 3);
  if (!f) return Promise.resolve({ irregulares: [] });
  bases['produtos_irregulares'] = 1;
  return D.busca('indices/irregulares_cnpj/' + f + '.json').then(function (r) {
    if (r.semRede) frios.push(r);
    var processos = (r.dados || {})[cnpj] || [];
    return Promise.all(processos.slice(0, 80).map(function (proc) {
      return irregularPorProcesso(proc, bases, frios);
    })).then(function (rs) {
      var itens = [];
      rs.forEach(function (x) { (x && x.irregulares || []).forEach(function (i) { itens.push(i); }); });
      return { irregulares: itens };
    });
  }).catch(function () { return { irregulares: [] }; });
}

function irregularPorProcesso(proc, bases, frios) {
  var f = D.CHAVE.irregulares(proc);
  if (!f) return Promise.resolve({ irregulares: [] });
  bases['produtos_irregulares'] = 1;
  return D.busca('produtos_irregulares/' + f + '.json').then(function (r) {
    if (r.semRede) frios.push(r);
    var d = (r.dados || {})[proc];
    return { irregulares: d ? [Object.assign({ processo: proc }, d)] : [] };
  }).catch(function () { return { irregulares: [] }; });
}
